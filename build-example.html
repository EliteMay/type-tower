STEP-BY-STEP BUILD EXAMPLE

HTMLだけじゃなく、
そのSTEPで必要なCSSも同じ場所に置く。

ロードマップのSTEP順に、どのファイルへ何を書くかをまとめています。HTML・CSS・JavaScript・JSONを別々のページで探さず、そのSTEPで触るコードを同じ場所から書ける構成です。

STEPごとにHTML + CSS + JS CSSはその場で追記 置換する処理は明記 最後に完成ファイル版と比較
1 HOMEと3つの塔 2 漢字問題 3 入力判定 4 階数 5 RESULT 6 TIME / COMBO 7 残り2つの塔 8 記録保存 9 見た目・階移動 10 テスト

HOW TO USE

上から順番に書く。古い関数は二重に残さない。

このページは完成コードを一気に貼るためではなく、STEPごとに動作確認しながら増やすための例です。同じ関数が後のSTEPに出てきた場合、ラベルが「置換」なら古い関数を消して新しい例へ入れ替えます。

「新規」

まだないファイルを作り、その内容を書く。

「追記」

今あるファイルの末尾、または指定場所へ追加する。

「置換」

前STEPの同名関数・同じ仮HTMLを消して入れ替える。

CSSの見方：各STEPの「CSS」欄をそのSTEPで game/css/style.css に追加します。後ろのSTEPほど上書きが必要な場合があるため、基本はSTEP順に末尾へ追記します。

const answerForm = document.getElementById('answerForm');
const answerInput = document.getElementById('answerInput');
const judgeMessage = document.getElementById('judgeMessage');

answerForm.addEventListener('submit', event => {
  event.preventDefault();

  const answer = normalizeAnswer(answerInput.value);
  const correct = normalizeAnswer(currentQuestion.answer);

  if (!answer) return;

  if (answer === correct) {
    handleCorrect();
  } else {
    handleMiss();
  }

  answerInput.value = '';
  showNextQuestion();
  answerInput.focus();
});

function normalizeAnswer(value) {
  return value.trim().toLowerCase().replace(/\s+/g, ' ');
}

// STEP 4で階数処理に置き換える仮関数

let floor = 1;

function handleCorrect() {

function updateFloorUI() {

      Number(item.dataset.floor) === floor

let correctCount = 0;
let missCount = 0;
let selectedMode = 'kanji';

async function startGame(mode) {
  selectedMode = mode;
  floor = 1;
  correctCount = 0;
  missCount = 0;
  updateFloorUI();
  showScreen('game');

  await loadQuestions();
  answerInput.focus();
}

// STEP 4のhandleCorrect / handleMissをこの内容へ更新
function handleCorrect() {
  correctCount += 1;
  floor += 1;
  updateFloorUI();

  if (floor >= 10) {
    finishGame();
    return;
  }
}

function handleMiss() {
  missCount += 1;
  floor = Math.max(1, floor - 1);
  updateFloorUI();
}

function finishGame() {
  document.getElementById('resultCorrect').textContent = correctCount;
  document.getElementById('resultMiss').textContent = missCount;
  showScreen('result');
}

function showNextQuestion() {
  const filtered = questions.filter(question => {
    return question.difficulty === selectedDifficulty;
  });

  const pool = filtered.length ? filtered : questions;
  const index = Math.floor(Math.random() * pool.length);
  currentQuestion = pool[index];

  answerInput.value = '';

  startQuestionTimer();
}

function startQuestionTimer() {
  clearInterval(timerId);
  timeLeft = TIME_BY_DIFFICULTY[selectedDifficulty] || 10;
  updateTimerUI();

  timerId = setInterval(() => {
    timeLeft -= 1;
    updateTimerUI();

    if (timeLeft <= 0) {
      clearInterval(timerId);
      handleMiss();
      showNextQuestion();
    }
  }, 1000);
}

const RECORD_KEY = 'typeTowerRecordsV1';

function loadRecords() {
  try {
    return JSON.parse(localStorage.getItem(RECORD_KEY)) || {};
  } catch (error) {
    console.warn('記録を読み込めませんでした', error);
    return {};
  }
}

function saveRecord(mode, difficulty, result) {
  const records = loadRecords();
  const key = `${mode}:${difficulty}`;
  const old = records[key] || {};

  const next = {
    bestTime: old.bestTime == null ? result.time : Math.min(old.bestTime, result.time),
    maxCombo: Math.max(old.maxCombo || 0, result.maxCombo),
    bestAccuracy: Math.max(old.bestAccuracy || 0, result.accuracy)
  };

  records[key] = next;

  try {
    localStorage.setItem(RECORD_KEY, JSON.stringify(records));
  } catch (error) {
    console.warn('記録を保存できませんでした', error);
  }

  return next;
}

function finishGame() {
  clearInterval(timerId);

  const elapsedSeconds = (performance.now() - startedAt) / 1000;
  const answered = correctCount + missCount;
  const accuracy = answered === 0
    ? 0
    : Math.round((correctCount / answered) * 100);

  document.getElementById('resultTitle').textContent = `${MODE_LABELS[selectedMode]} CLEAR`;
  document.getElementById('resultCorrect').textContent = correctCount;
  document.getElementById('resultMiss').textContent = missCount;
  document.getElementById('resultAccuracy').textContent = `${accuracy}%`;
  document.getElementById('resultMaxCombo').textContent = maxCombo;
  document.getElementById('resultTime').textContent = `${elapsedSeconds.toFixed(1)}s`;

  const best = saveRecord(selectedMode, selectedDifficulty, {
    time: elapsedSeconds,
    maxCombo,
    accuracy
  });

  document.getElementById('bestRecordText').textContent =
    `BEST TIME ${best.bestTime.toFixed(1)}s / MAX COMBO ${best.maxCombo} / ACC ${best.bestAccuracy}%`;

  showScreen('result');
}
