そのSTEPで書くHTML・CSS・JSを、

1 HOMEと3つの塔 2 漢字問題 3 入力判定 4 階数 5 RESULT 6 TIME / COMBO 7 残り2つの塔 8 記録保存 9 見た目・階移動 10 テスト

HOW TO USE

STEP順に書く。置換と書いてある古いコードは残さない。

後のSTEPで同じ関数を強化するため、古い関数と新しい関数を両方残すとエラーになります。「置換」は前のコードを消して新しい例に入れ替えてください。

新規

まだないファイルを作り、コードを書く。

追記

指定ファイルへ追加する。CSSは基本的に末尾へ追加。

置換

前STEPの同名関数・仮HTMLを消して入れ替える。

CSSは同じSTEP内：そのSTEPで増やしたHTMLのすぐ下に、対応する game/css/style.css の追記例を置いています。

const answerForm = document.getElementById('answerForm');
const answerInput = document.getElementById('answerInput');
const judgeMessage = document.getElementById('judgeMessage');

answerForm.addEventListener('submit', async event => {
  event.preventDefault();

  const answer = normalizeAnswer(answerInput.value);
  if (!answer || !currentQuestion) return;

  if (answer === normalizeAnswer(currentQuestion.answer)) {
    await handleCorrect();
  } else {
    await handleMiss();
  }
});

function normalizeAnswer(value) {
  return value.trim().toLowerCase().replace(/\s+/g, ' ');
}

async function handleCorrect() {
  judgeMessage.textContent = '正解！';
  answerInput.value = '';
  showNextQuestion();
  answerInput.focus();
}

async function handleMiss() {
  judgeMessage.textContent = 'MISS';
  answerInput.value = '';
  showNextQuestion();
  answerInput.focus();
}

function showNextQuestion() {
  const filtered = questions.filter(q => q.difficulty === selectedDifficulty);
  const pool = filtered.length ? filtered : questions;
  currentQuestion = pool[Math.floor(Math.random() * pool.length)];

const RECORD_KEY = 'typeTowerRecordsV1';

function loadRecords() {
  try {
    return JSON.parse(localStorage.getItem(RECORD_KEY)) || {};
  } catch (error) {
    console.warn('記録を読み込めませんでした', error);
    return {};
  }
}

function saveRecord(mode, difficulty, result) {
  const records = loadRecords();
  const key = `${mode}:${difficulty}`;
  const old = records[key] || {};
  const next = {
    bestTime: old.bestTime == null ? result.time : Math.min(old.bestTime, result.time),
    maxCombo: Math.max(old.maxCombo || 0, result.maxCombo),
    bestAccuracy: Math.max(old.bestAccuracy || 0, result.accuracy)
  };
  records[key] = next;
  try {
    localStorage.setItem(RECORD_KEY, JSON.stringify(records));
  } catch (error) {
    console.warn('記録を保存できませんでした', error);
  }
  return next;
}

}
