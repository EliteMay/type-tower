FULL BUILD EXAMPLE

このページを上から作れば、
最低限遊べるTYPE TOWERになる。

断片的なコード例ではなく、どのファイルに何を書くかを一式で載せます。最初は画像素材がなくても動く形にし、3つの塔・3モード・階数・TIME・COMBO・RESULTまでつながる例です。

HOME：空 + 3つの塔 GAME：1F → 10F 正解 +1F / MISS -1F 3モード TIME / COMBO RESULT / 記録

HOW TO USE

一気に全部書かず、1ファイルずつ動かす

下の例は完成形に近いですが、授業では「HTML → CSS → main.js → game.js → JSON」の順で追加し、各段階でブラウザ確認する方が壊れた場所を見つけやすいです。

最初

HTMLとCSSだけでHOMEに3つの塔が見えるところまで。

次

漢字の塔だけGAMEにつないで、問題表示・判定・階数を確認。

最後

英訳・和訳、TIME、COMBO、保存、演出を足す。

const screens = [...document.querySelectorAll('[data-screen]')];

function showScreen(name) {
  screens.forEach(screen => {
    screen.hidden = screen.dataset.screen !== name;
  });
}

document.querySelectorAll('[data-mode]').forEach(button => {
  button.addEventListener('click', () => {
    startGame(button.dataset.mode);
  });
});

const RECORD_KEY = 'typeTowerRecordsV1';

function loadRecords() {
  try {
    return JSON.parse(localStorage.getItem(RECORD_KEY)) || {};
  } catch (error) {
    console.warn('記録を読み込めませんでした', error);
    return {};
  }
}

function saveRecord(mode, difficulty, result) {
  const records = loadRecords();
  const key = `${mode}:${difficulty}`;
  const old = records[key] || {};

  const next = {
    bestTime: old.bestTime == null
      ? result.time
      : Math.min(old.bestTime, result.time),
    maxCombo: Math.max(old.maxCombo || 0, result.maxCombo),
    bestAccuracy: Math.max(old.bestAccuracy || 0, result.accuracy)
  };

  records[key] = next;

  try {
    localStorage.setItem(RECORD_KEY, JSON.stringify(records));
  } catch (error) {
    console.warn('記録を保存できませんでした', error);
  }

  return next;
}

async function startGame(mode) {
  cancelGame();

  selectedMode = mode;
  selectedDifficulty = document.getElementById('difficultySelect').value;
  floor = 1;
  combo = 0;
  maxCombo = 0;
  correctCount = 0;

function cancelGame() {
  clearInterval(timerId);
  timerId = null;
  questionLocked = true;
}

async function loadModeQuestions() {
  try {
    const response = await fetch(DATA_FILES[selectedMode]);
    if (!response.ok) throw new Error(`HTTP ${response.status}`);

    questions = await response.json();

    if (!Array.isArray(questions) || questions.length === 0) {
      throw new Error('問題データが空です');
    }

    return true;
  } catch (error) {
    console.error('問題を読み込めませんでした', error);
    document.getElementById('questionText').textContent = '問題を読み込めませんでした';
    return false;
  }
}

function showNextQuestion() {
  questionLocked = false;

  const filtered = questions.filter(question => {
    return question.difficulty === selectedDifficulty;
  });

  const pool = filtered.length ? filtered : questions;
  const index = Math.floor(Math.random() * pool.length);
  currentQuestion = pool[index];

  document.getElementById('questionText').textContent = currentQuestion.question;
  answerInput.value = '';
  answerInput.focus();

  startQuestionTimer();
}

answerForm.addEventListener('submit', async event => {
  event.preventDefault();

  if (questionLocked || !currentQuestion) return;

  const answer = answerInput.value.trim();
  if (!answer) return;

  if (normalizeAnswer(answer) === normalizeAnswer(currentQuestion.answer)) {
    await handleCorrect();
  } else {
    await handleMiss();
  }
});

function normalizeAnswer(value) {
  return value
    .trim()
    .toLowerCase()
    .replace(/\s+/g, ' ');
}

async function handleCorrect() {
  if (questionLocked) return;
  questionLocked = true;
  clearInterval(timerId);

  correctCount += 1;
  combo += 1;
  maxCombo = Math.max(maxCombo, combo);
  floor += 1;

  updateComboUI();
  updateFloorUI();

  await Promise.all([
    flashAnswer('correct'),
    playFloorMove('up')
  ]);

  if (floor >= 10) {
    finishGame();
    return;
  }

  showNextQuestion();
}

async function handleMiss() {
  if (questionLocked) return;
  questionLocked = true;
  clearInterval(timerId);

  missCount += 1;
  combo = 0;
  floor = Math.max(1, floor - 1);

  updateComboUI();
  updateFloorUI();

  await Promise.all([
    flashAnswer('miss'),
    playFloorMove('down')
  ]);

  showNextQuestion();
}

function updateFloorUI() {
  document.getElementById('floorText').textContent = `${floor}F`;

  document.querySelectorAll('[data-floor]').forEach(item => {
    const isCurrent = Number(item.dataset.floor) === floor;
    item.classList.toggle('is-current', isCurrent);
  });
}

function updateComboUI() {
  document.getElementById('comboText').textContent = combo;
}

function startQuestionTimer() {
  clearInterval(timerId);

  timeLeft = TIME_BY_DIFFICULTY[selectedDifficulty] || 10;
  updateTimerUI();

  timerId = setInterval(async () => {
    timeLeft -= 1;
    updateTimerUI();

    if (timeLeft <= 0) {
      clearInterval(timerId);
      await handleMiss();
    }
  }, 1000);
}

function updateTimerUI() {
  document.getElementById('timeText').textContent = timeLeft;
}

function finishGame() {
  cancelGame();

  const elapsedSeconds = (performance.now() - startedAt) / 1000;
  const answered = correctCount + missCount;
  const accuracy = answered === 0
    ? 0
    : Math.round((correctCount / answered) * 100);

  document.getElementById('resultTitle').textContent = `${MODE_LABELS[selectedMode]} CLEAR`;
  document.getElementById('resultCorrect').textContent = correctCount;
  document.getElementById('resultMiss').textContent = missCount;
  document.getElementById('resultAccuracy').textContent = `${accuracy}%`;
  document.getElementById('resultMaxCombo').textContent = maxCombo;
  document.getElementById('resultTime').textContent = `${elapsedSeconds.toFixed(1)}s`;

  const best = saveRecord(selectedMode, selectedDifficulty, {
    time: elapsedSeconds,
    maxCombo,
    accuracy
  });

  document.getElementById('bestRecordText').textContent = best
    ? `BEST TIME ${best.bestTime.toFixed(1)}s / MAX COMBO ${best.maxCombo} / ACC ${best.bestAccuracy}%`
    : '';

  showScreen('result');
}
